"""Diagnostic positive control (not a registered experiment): plain photometric 3DGS
from random init in the camera-only bound, 12 train views, 4 held-out views."""
import math, sys, json, torch
from rtgs.data.calibrated import load_calibrated_scene, _object_bounds
from rtgs.core.gaussians3d import Gaussians3D
from rtgs.render.gsplat_backend import GsplatRasterizer
root="dataset/external/haelyn_halfbody/prepared"
train=["C0001","C0002","C0003","C0005","C0006","C0007","C0009","C0010","C0011","C0013","C0014","C0015"]
held=["C0004","C0008","C0012","C0016"]
def load(ids):
    s=load_calibrated_scene(root, calibration_path=f"{root}/calibration_dome.json", downscale=1, view_ids=ids, load_masks=True)
    return [c.to("cuda") for c in s.cameras],[i.cuda() for i in s.images],[m.cuda().squeeze().bool() for m in s.masks]
tc,ti,_=load(train); hc,hi,hm=load(held)
center,extent=_object_bounds(tc,None); center=torch.as_tensor(center).float().cuda()
r=GsplatRasterizer(packed=False, antialiased=False)
def run(N, iters, seed=0):
    g=torch.Generator(device="cuda").manual_seed(seed)
    d=torch.randn(N,3,device="cuda",generator=g); d=d/d.norm(dim=1,keepdim=True)*torch.rand(N,1,device="cuda",generator=g)**(1/3)
    p={"means":(center+0.6*extent*d),"quats":torch.nn.functional.normalize(torch.randn(N,4,device="cuda",generator=g),dim=1),
       "log_scales":torch.full((N,3),math.log(0.6*extent/N**(1/3)),device="cuda"),
       "opa":torch.full((N,),-2.0,device="cuda"),"sh":torch.zeros(N,4,3,device="cuda")}
    for v in p.values(): v.requires_grad_(True)
    lrs={"means":1.6e-4*extent*10,"quats":1e-3,"log_scales":5e-3,"opa":5e-2,"sh":2.5e-3*4}
    opt=torch.optim.Adam([{"params":[p[k]],"lr":lrs[k]} for k in p],eps=1e-15)
    def model(): return Gaussians3D(p["means"],torch.nn.functional.normalize(p["quats"],dim=1),p["log_scales"],torch.sigmoid(p["opa"]),p["sh"])
    bg=torch.zeros(3,device="cuda")
    for it in range(iters):
        k=it%len(tc); out=r.render(model(),tc[k],background=bg)
        loss=(out.color-ti[k]).abs().mean()*0.8+((out.color-ti[k])**2).mean()*0.2
        opt.zero_grad(); loss.backward(); opt.step()
    res={}
    with torch.no_grad():
        for name,C,I,M in (("train",tc,ti,None),("held",hc,hi,hm)):
            ps=[];fg=[]
            for j,(c,im) in enumerate(zip(C,I)):
                o=r.render(model(),c,background=bg).color.clamp(0,1)
                ps.append(-10*math.log10(((o-im)**2).mean().item()))
                if M is not None:
                    mm=M[j].reshape(o.shape[0],o.shape[1]).float()[...,None]; fg.append(-10*math.log10((((o-im)**2)*mm).sum().item()/(3*mm.sum().item())))
            res[name]=sum(ps)/len(ps)
            if fg: res["held_fg"]=sum(fg)/len(fg)
    return res
for N,iters in [(256,3000),(256,15000),(20000,15000)]:
    print(N,iters,json.dumps({k:round(v,2) for k,v in run(N,iters).items()}),flush=True)
